# ============================================================
# Directory
# ============================================================

BIN_DIR   := bin
SRC_DIR   := src
BUILD_DIR := build
INC_DIR   := include

# ============================================================
# Source directories
# ============================================================

SRC01_DIR   := $(SRC_DIR)/01_CATCH_HDPRM_TDC
SRC02_DIR   := $(SRC_DIR)/02_CFT_PedCor
SRC03_1_DIR := $(SRC_DIR)/03_CFT_PosCor/01_PHI
SRC03_2_DIR := $(SRC_DIR)/03_CFT_PosCor/02_UV
SRC04_DIR   := $(SRC_DIR)/04_CFT_PHC
SRC05_DIR   := $(SRC_DIR)/05_CATCH_pid
SRC06_DIR   := $(SRC_DIR)/06_CFT_Mip_particle_select
SRC07_DIR   := $(SRC_DIR)/07_BGO_EneCalib_comp
SRC08_DIR   := $(SRC_DIR)/08_CFT_EneCalib

SRC_DIRS := \
	$(SRC01_DIR) \
	$(SRC02_DIR) \
	$(SRC03_1_DIR) \
	$(SRC03_2_DIR) \
	$(SRC04_DIR) \
	$(SRC05_DIR) \
	$(SRC06_DIR) \
	$(SRC07_DIR) \
	$(SRC08_DIR)

# ============================================================
# Source files
# ============================================================

SRC01   := $(wildcard $(SRC01_DIR)/*.cc)
SRC02   := $(wildcard $(SRC02_DIR)/*.cc)
SRC03_1 := $(wildcard $(SRC03_1_DIR)/*.cc)
SRC03_2 := $(wildcard $(SRC03_2_DIR)/*.cc)
SRC04   := $(wildcard $(SRC04_DIR)/*.cc)
SRC05   := $(wildcard $(SRC05_DIR)/*.cc)
SRC06   := $(wildcard $(SRC06_DIR)/*.cc)
SRC07   := $(wildcard $(SRC07_DIR)/*.cc)
SRC08   := $(wildcard $(SRC08_DIR)/*.cc)

ALL_SRCS := \
	$(SRC01) $(SRC02) \
	$(SRC03_1) $(SRC03_2) \
	$(SRC04) $(SRC05) $(SRC06) \
	$(SRC07) $(SRC08)

# ============================================================
# Object files and executables
# ============================================================

# Preserve source directory structure under build/
objects = $(addprefix $(BUILD_DIR)/,$(patsubst $(SRC_DIR)/%,%,$(1:.cc=.o)))

# Executables are placed directly under bin/
targets = $(addprefix $(BIN_DIR)/,$(notdir $(basename $(1))))

OBJ01   := $(call objects,$(SRC01))
OBJ02   := $(call objects,$(SRC02))
OBJ03_1 := $(call objects,$(SRC03_1))
OBJ03_2 := $(call objects,$(SRC03_2))
OBJ04   := $(call objects,$(SRC04))
OBJ05   := $(call objects,$(SRC05))
OBJ06   := $(call objects,$(SRC06))
OBJ07   := $(call objects,$(SRC07))
OBJ08   := $(call objects,$(SRC08))

ALL_OBJS := $(call objects,$(ALL_SRCS))
ALL_BINS := $(call targets,$(ALL_SRCS))
DEPFILES := $(ALL_OBJS:.o=.d)

# ============================================================
# Compiler
# ============================================================

CXX := /usr/bin/g++

CPPFLAGS += -Dlynx
CPPFLAGS += -DLINUX
CPPFLAGS += -DOFFLINE
CPPFLAGS += -I$(INC_DIR)

CXXFLAGS += -O2
CXXFLAGS += -g
CXXFLAGS += -std=c++17
CXXFLAGS += -m64
CXXFLAGS += $(OPT)

# ============================================================
# ROOT
# ============================================================

USE_ROOT ?= 1
ROOT_CONFIG ?= root-config

ifeq ($(USE_ROOT),1)

ROOT_CFLAGS := $(shell $(ROOT_CONFIG) --cflags)
ROOT_LIBS   := $(shell $(ROOT_CONFIG) --libs)

CPPFLAGS += -DUSE_ROOT
CXXFLAGS += $(ROOT_CFLAGS)
LDLIBS   += $(ROOT_LIBS) -lEG

endif

# ============================================================
# Libraries
# ============================================================

LDLIBS += -L/usr/X11R6/lib
LDLIBS += -lX11 -lnsl -lcrypt -ldl -lc -lm

# ============================================================
# Build rules
# ============================================================

.PHONY: all clean cleanhist \
	usr01 usr02 usr03 usr03_1 usr03_2 \
	usr04 usr05 usr06 usr07 usr08

all: usr01 usr02 usr03 usr04 usr05 usr06 usr07 usr08

usr01: $(call targets,$(SRC01))
usr02: $(call targets,$(SRC02))
usr03: usr03_1 usr03_2
usr03_1: $(call targets,$(SRC03_1))
usr03_2: $(call targets,$(SRC03_2))
usr04: $(call targets,$(SRC04))
usr05: $(call targets,$(SRC05))
usr06: $(call targets,$(SRC06))
usr07: $(call targets,$(SRC07))
usr08: $(call targets,$(SRC08))

# ------------------------------------------------------------
# Compile
# ------------------------------------------------------------

$(BUILD_DIR)/%.o: $(SRC_DIR)/%.cc
	@mkdir -p $(@D)
	$(CXX) $(CPPFLAGS) $(CXXFLAGS) \
		-MMD -MP -MF $(@:.o=.d) \
		-c $< -o $@

# ------------------------------------------------------------
# Link
# ------------------------------------------------------------

define PROGRAM_template

$(BIN_DIR)/$(notdir $(basename $(1))): \
	$(BUILD_DIR)/$(patsubst $(SRC_DIR)/%,%,$(1:.cc=.o))
	@mkdir -p $$(@D)
	$(CXX) -o $$@ $$^ $(LDFLAGS) $(LDLIBS)

endef

$(foreach src,$(ALL_SRCS),$(eval $(call PROGRAM_template,$(src))))

# ------------------------------------------------------------
# Header dependencies
# ------------------------------------------------------------

-include $(DEPFILES)

# ============================================================
# Clean
# ============================================================

clean:
	rm -f $(ALL_OBJS) $(DEPFILES) $(ALL_BINS)
	rm -f core $(BUILD_DIR)/*.o *~

cleanhist:
	rm -f core *.hbk *.root
